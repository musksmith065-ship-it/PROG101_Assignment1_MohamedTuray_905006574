Function Main
    ... MediQueue SL - triage and queue management Mohamed Turay - 905006574 (PROG101 Assignment 1)
    ... Constants (Flowgorithm has no constant statement, so they are set once here)
    Declare Integer MaxQueueSize
    Declare Integer FairnessLimit
    Declare Integer AvgConsultMinutes
    Declare Integer choice
    Declare Integer i
    Declare Boolean running

    ... stats: 0 eCount, 1 uCount, 2 rCount, 3 totalRegistered, 4 servedCount, 5 consecutiveUrgent, 6 eTotal, 7 uTotal, 8 rTotal, 9 infectiousCount
    Declare Integer Array stats[10]
    Declare String Array eName[200]
    Declare String Array uName[200]
    Declare String Array rName[200]
    Declare Integer Array eTicket[200]
    Declare Integer Array uTicket[200]
    Declare Integer Array rTicket[200]

    Assign MaxQueueSize = 200
    Assign FairnessLimit = 4
    Assign AvgConsultMinutes = 10
    For i = 0 to 9
        Assign stats[i] = 0
    End
    Assign running = true
    While running == true
        Call DisplayMenu()
        Assign choice = GetValidatedInteger("Enter choice (1-5)", 1, 5)
        If choice == 1
            Call RegisterPatient(eName, eTicket, uName, uTicket, rName, rTicket, stats, MaxQueueSize, AvgConsultMinutes)
        Else
            If choice == 2
                Call CallNextPatient(eName, eTicket, uName, uTicket, rName, rTicket, stats, FairnessLimit)
            Else
                If choice == 3
                    Call ShowQueueStatus(eName, uName, rName, stats)
                Else
                    If choice == 4
                        Call GenerateDailySummary(stats)
                    Else
                        Assign running = false
                    End
                End
            End
        End
    End
    Output "System closed. Stay safe."
End

Function AddToQueue (Integer priority, String patientName, Integer ticketNo, String Array eName, Integer Array eTicket, String Array uName, Integer Array uTicket, String Array rName, Integer Array rTicket, Integer Array stats)
    If priority == 1
        Assign eName[stats[0]] = patientName
        Assign eTicket[stats[0]] = ticketNo
        Assign stats[0] = stats[0] + 1
        Assign stats[6] = stats[6] + 1
    Else
        If priority == 2
            Assign uName[stats[1]] = patientName
            Assign uTicket[stats[1]] = ticketNo
            Assign stats[1] = stats[1] + 1
            Assign stats[7] = stats[7] + 1
        Else
            Assign rName[stats[2]] = patientName
            Assign rTicket[stats[2]] = ticketNo
            Assign stats[2] = stats[2] + 1
            Assign stats[8] = stats[8] + 1
        End
    End
    Assign stats[3] = stats[3] + 1
End

Function AssignPriority (Integer age, Real temperature, Integer days, Boolean unconscious, Boolean severeBleeding, Boolean breathingDifficulty, Boolean pregnant, Boolean dangerSign)
    ... Constants: CHILD_AGE_LIMIT 5, ELDERLY_AGE_LIMIT 65, FEVER_LIMIT 38.0, HIGH_FEVER_LIMIT 39.5, LONG_ILLNESS_DAYS 3
    Declare Boolean isEmergency
    Declare Boolean isUrgent
    Declare Integer result

    Assign isEmergency = unconscious OR severeBleeding OR breathingDifficulty OR (pregnant AND dangerSign) OR (age < 5 AND temperature >= 39.5)
    Assign isUrgent = temperature >= 38.0 OR pregnant OR age < 5 OR age >= 65 OR days >= 3
    If isEmergency
        Assign result = 1
    Else
        If isUrgent
            Assign result = 2
        Else
            Assign result = 3
        End
    End
    Return Integer result
End

Function CallNextPatient (String Array eName, Integer Array eTicket, String Array uName, Integer Array uTicket, String Array rName, Integer Array rTicket, Integer Array stats, Integer fairnessLimit)
    Declare Integer i
    Declare Integer calledTicket
    Declare Integer calledPriority
    Declare String calledName

    If stats[0] + stats[1] + stats[2] == 0
        Output "No patients are waiting."
    Else
        If stats[0] > 0
            Assign calledName = eName[0]
            Assign calledTicket = eTicket[0]
            For i = 0 to stats[0] - 2
                Assign eName[i] = eName[i + 1]
                Assign eTicket[i] = eTicket[i + 1]
            End
            Assign stats[0] = stats[0] - 1
            Assign calledPriority = 1
        Else
            If stats[1] > 0 AND NOT (stats[5] >= fairnessLimit AND stats[2] > 0)
                Assign calledName = uName[0]
                Assign calledTicket = uTicket[0]
                For i = 0 to stats[1] - 2
                    Assign uName[i] = uName[i + 1]
                    Assign uTicket[i] = uTicket[i + 1]
                End
                Assign stats[1] = stats[1] - 1
                Assign calledPriority = 2
                Assign stats[5] = stats[5] + 1
            Else
                Assign calledName = rName[0]
                Assign calledTicket = rTicket[0]
                For i = 0 to stats[2] - 2
                    Assign rName[i] = rName[i + 1]
                    Assign rTicket[i] = rTicket[i + 1]
                End
                Assign stats[2] = stats[2] - 1
                Assign calledPriority = 3
                Assign stats[5] = 0
            End
        End
        Assign stats[4] = stats[4] + 1
        Output "Now calling: " & calledName & " | " & ColourName(calledPriority) & " | Ticket " & calledTicket
    End
End

Function ColourName (Integer priority)
    Declare String colour

    If priority == 1
        Assign colour = "RED"
    Else
        If priority == 2
            Assign colour = "YELLOW"
        Else
            Assign colour = "GREEN"
        End
    End
    Return String colour
End

Function DisplayMenu
    Output "========== MEDIQUEUE SL =========="
    Output "1 Register patient"
    Output "2 Call next patient"
    Output "3 Show queue status"
    Output "4 Daily summary"
    Output "5 Exit"
End

Function EstimateWaitTime (Integer priority, Integer Array stats, Integer avgMinutes)
    Declare Integer ahead
    Declare Integer result

    If priority == 1
        Assign ahead = stats[0] - 1
    Else
        If priority == 2
            Assign ahead = stats[0] + stats[1] - 1
        Else
            Assign ahead = stats[0] + stats[1] + stats[2] - 1
        End
    End
    Assign result = ahead * avgMinutes
    Return Integer result
End

Function GenerateDailySummary (Integer Array stats)
    Declare Real pct

    If stats[3] == 0
        Output "No patients registered today."
    Else
        Output "Registered: " & stats[3] & "   Served: " & stats[4]
        Assign pct = stats[6] * 100.0 / stats[3]
        Output "Emergency: " & stats[6] & " (" & Int(pct * 10 + 0.5) / 10 & "%)"
        Assign pct = stats[7] * 100.0 / stats[3]
        Output "Urgent: " & stats[7] & " (" & Int(pct * 10 + 0.5) / 10 & "%)"
        Assign pct = stats[8] * 100.0 / stats[3]
        Output "Routine: " & stats[8] & " (" & Int(pct * 10 + 0.5) / 10 & "%)"
        Output "Suspected infectious cases: " & stats[9]
    End
End

Function GetValidatedInteger (String prompt, Integer low, Integer high)
    Declare Integer value

    Do
        Output prompt
        Input value
        If value < low OR value > high
            Output "Invalid. Enter " & low & " to " & high
        End
    Loop value < low OR value > high
    Return Integer value
End

Function GetValidatedReal (String prompt, Real low, Real high)
    Declare Real value

    Do
        Output prompt
        Input value
        If value < low OR value > high
            Output "Invalid. Enter " & low & " to " & high
        End
    Loop value < low OR value > high
    Return Real value
End

Function GetYesNo (String prompt)
    Declare String answer
    Declare Boolean valid
    Declare Boolean result

    Do
        Output prompt & " (Y/N)"
        Input answer
        Assign valid = answer == "Y" OR answer == "y" OR answer == "N" OR answer == "n"
        If valid == false
            Output "Please enter Y or N"
        End
    Loop valid == false
    Assign result = answer == "Y" OR answer == "y"
    Return Boolean result
End

Function RegisterPatient (String Array eName, Integer Array eTicket, String Array uName, Integer Array uTicket, String Array rName, Integer Array rTicket, Integer Array stats, Integer maxQueue, Integer avgMinutes)
    Declare String patientName
    Declare String sex
    Declare Integer age
    Declare Integer days
    Declare Integer priority
    Declare Integer ticketNo
    Declare Integer waitMinutes
    Declare Real temperature
    Declare Boolean unconscious
    Declare Boolean severeBleeding
    Declare Boolean breathingDifficulty
    Declare Boolean suspectedInfectious
    Declare Boolean pregnant
    Declare Boolean dangerSign
    Declare Boolean queueFull

    Assign ticketNo = stats[3] + 1
    Output "Patient name:"
    Input patientName
    Output "Sex (F or M):"
    Input sex
    Assign age = GetValidatedInteger("Age in years (0-120)", 0, 120)
    Assign temperature = GetValidatedReal("Temperature in C (30.0-45.0)", 30.0, 45.0)
    Assign days = GetValidatedInteger("Days ill (0-365)", 0, 365)
    Assign unconscious = GetYesNo("Unconscious or convulsing?")
    Assign severeBleeding = GetYesNo("Severe bleeding?")
    Assign breathingDifficulty = GetYesNo("Difficulty breathing?")
    Assign suspectedInfectious = GetYesNo("Signs of cholera, Lassa fever or Ebola?")
    If (sex == "F" OR sex == "f") AND age >= 12 AND age <= 50
        Assign pregnant = GetYesNo("Pregnant?")
        If pregnant == true
            Assign dangerSign = GetYesNo("Pregnancy danger sign?")
        Else
            Assign dangerSign = false
        End
    Else
        Assign pregnant = false
        Assign dangerSign = false
    End
    Assign priority = AssignPriority(age, temperature, days, unconscious, severeBleeding, breathingDifficulty, pregnant, dangerSign)
    Assign queueFull = (priority == 1 AND stats[0] >= maxQueue) OR (priority == 2 AND stats[1] >= maxQueue) OR (priority == 3 AND stats[2] >= maxQueue)
    If queueFull == true
        Output "Queue full. Refer patient to the nearest CHC."
    Else
        Call AddToQueue(priority, patientName, ticketNo, eName, eTicket, uName, uTicket, rName, rTicket, stats)
        If suspectedInfectious == true
            Assign stats[9] = stats[9] + 1
            Output "ALERT: isolate patient and inform the nurse in charge."
        End
        Assign waitMinutes = EstimateWaitTime(priority, stats, avgMinutes)
        Output "============ MEDIQUEUE SL TICKET ============"
        Output "Ticket No: " & ticketNo & "   Name: " & patientName
        Output "Priority: " & ColourName(priority)
        Output "Estimated wait: " & waitMinutes & " minutes"
        Output "============================================="
    End
End

Function ShowQueueStatus (String Array eName, String Array uName, String Array rName, Integer Array stats)
    Declare Integer i

    Output "EMERGENCY (RED): " & stats[0]
    For i = 0 to stats[0] - 1
        Output (i + 1) & ". " & eName[i]
    End
    Output "URGENT (YELLOW): " & stats[1]
    For i = 0 to stats[1] - 1
        Output (i + 1) & ". " & uName[i]
    End
    Output "ROUTINE (GREEN): " & stats[2]
    For i = 0 to stats[2] - 1
        Output (i + 1) & ". " & rName[i]
    End
End
